## Basic Setup

Let's create a simple todo application to demonstrate kSync's powerful new features including database-like queries, CRDTs, and React integration.

### 1. Define Your Data Schema

First, define the structure of your data using Zod schemas:

```typescript
import { z } from 'zod';
import { ksync } from '@klastra/ksync';

// Define data schemas
const TodoSchema = z.object({
  id: z.string(),
  title: z.string(),
  completed: z.boolean(),
  priority: z.enum(['low', 'medium', 'high']),
  createdAt: z.number(),
  updatedAt: z.number(),
  userId: z.string()
});

// Create kSync with Drizzle adapter for database-like queries
const todoApp = ksync.createDrizzleAdapter({
  schema: {
    todos: TodoSchema
  }
});

// Define event schemas for mutations
ksync.defineSchema('todo-created', TodoSchema);
ksync.defineSchema('todo-updated', TodoSchema.partial().extend({ id: z.string() }));
ksync.defineSchema('todo-deleted', z.object({ id: z.string() }));
```

### 2. Database-Like Queries

Query your real-time data with familiar SQL-like syntax:

```typescript
// Find all todos
const allTodos = await todoApp.todos.findMany();

// Find active todos
const activeTodos = await todoApp.todos.findMany({
  where: { completed: false },
  orderBy: { priority: 'desc', createdAt: 'desc' }
});

// Find high priority todos
const highPriorityTodos = await todoApp.todos.findMany({
  where: { 
    priority: 'high',
    completed: false 
  },
  limit: 10
});

// Search todos
const searchResults = await todoApp.todos.findMany({
  where: {
    title: { contains: 'project' }
  }
});

console.log('Active todos:', activeTodos);
```

### 3. Create and Update Data

Use events to create and update data:

```typescript
// Create a new todo
async function createTodo(title: string, priority: 'low' | 'medium' | 'high' = 'medium') {
  const todo = {
    id: `todo-${Date.now()}`,
    title,
    completed: false,
    priority,
    createdAt: Date.now(),
    updatedAt: Date.now(),
    userId: 'user-123'
  };
  
  await ksync.send('todo-created', todo);
  return todo;
}

// Update a todo
async function updateTodo(id: string, updates: Partial<TodoSchema>) {
  await ksync.send('todo-updated', {
    id,
    ...updates,
    updatedAt: Date.now()
  });
}

// Delete a todo
async function deleteTodo(id: string) {
  await ksync.send('todo-deleted', { id });
}

// Usage
await createTodo('Buy groceries', 'high');
await createTodo('Read documentation', 'medium');
await updateTodo('todo-123', { completed: true });
```

### 4. React Integration

Use React hooks for seamless real-time updates:

```tsx
import React from 'react';
import { KSyncProvider, useKSyncLiveQuery, useKSync } from '@klastra/ksync/react';

function TodoApp() {
  return (
    <KSyncProvider instance={ksync}>
      <TodoList />
      <AddTodoForm />
    </KSyncProvider>
  );
}

function TodoList() {
  // Live query that automatically updates
  const { data: todos, loading } = useKSyncLiveQuery(
    async () => {
      return await todoApp.todos.findMany({
        where: { completed: false },
        orderBy: { priority: 'desc', createdAt: 'desc' }
      });
    },
    [] // Dependencies
  );

  if (loading) return <div>Loading todos...</div>;

  return (
    <ul>
      {todos.map(todo => (
        <TodoItem key={todo.id} todo={todo} />
      ))}
    </ul>
  );
}

function TodoItem({ todo }) {
  const ksync = useKSync();
  
  const toggleComplete = async () => {
    await ksync.send('todo-updated', {
      id: todo.id,
      completed: !todo.completed,
      updatedAt: Date.now()
    });
  };

  return (
    <li onClick={toggleComplete} style={{ 
      textDecoration: todo.completed ? 'line-through' : 'none',
      color: todo.priority === 'high' ? 'red' : 'black'
    }}>
      {todo.title}
    </li>
  );
}

function AddTodoForm() {
  const ksync = useKSync();
  const [title, setTitle] = React.useState('');

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!title.trim()) return;

    await ksync.send('todo-created', {
      id: `todo-${Date.now()}`,
      title,
      completed: false,
      priority: 'medium',
      createdAt: Date.now(),
      updatedAt: Date.now(),
      userId: 'user-123'
    });

    setTitle('');
  };

  return (
    <form onSubmit={handleSubmit}>
      <input 
        value={title}
        onChange={(e) => setTitle(e.target.value)}
        placeholder="Add a todo..."
      />
      <button type="submit">Add</button>
    </form>
  );
}
```

## Choose Your Sync Mode

kSync supports two powerful synchronization modes:

### Real-Time WebSocket Sync

Perfect for live collaboration and real-time updates:

```typescript
// Create with WebSocket sync
const realtimeSync = ksync.create({
  syncUrl: 'ws://localhost:3001',
  mode: 'websocket'
});

// Events sync automatically in real-time
await realtimeSync.send('todo-created', { /* todo data */ });
// ↑ This instantly syncs to all connected clients
```

### Git-Like Sync (Offline-First)

Perfect for offline-first apps with manual/scheduled sync:

```typescript
// Create with Git-like sync
const offlineSync = ksync.createGitSync({
  serverUrl: 'https://api.myapp.com/sync',
  auth: { token: 'your-auth-token' }
});

// Work offline
await offlineSync.send('todo-created', { /* todo data */ });
await offlineSync.send('todo-updated', { /* updates */ });

// Sync when ready (manual)
await offlineSync.sync(); // Pull + Push

// Or just pull/push separately
await offlineSync.pull(); // Download changes
await offlineSync.push(); // Upload changes

// Or enable periodic sync
offlineSync.startPeriodicSync({ interval: 60000 }); // Every minute
```

## Advanced Features

### CRDT for Conflict-Free Collaboration

Enable true collaborative editing without conflicts:

```typescript
// Create CRDT-enabled instance
const crdtSync = ksync.createCRDT({
  nodeId: 'user-alice',
  mode: 'event-sourcing'
});

// Create collaborative data structures
const documentTitle = crdtSync.createLWWRegister('doc-title', 'Untitled');
const documentTags = crdtSync.createGSet('doc-tags');
const viewCount = crdtSync.createGCounter('doc-views');

// Multiple users can edit simultaneously without conflicts
await documentTitle.set('My Amazing Document');
await documentTags.add('important');
await viewCount.increment();
```

### Multi-Store Architecture

Organize complex apps with multiple isolated stores:

```typescript
// Create multi-store manager
const multistore = ksync.createMultistore();

// Create separate stores for different domains
const userStore = await multistore.createStore('users', {
  schema: { user: UserSchema }
});

const projectStore = await multistore.createStore('projects', {
  schema: { project: ProjectSchema }
});

// Cross-store operations
await multistore.broadcast('user-status-changed', {
  userId: 'user-123',
  status: 'online'
});

// Derived stores
const activityStore = await multistore.createDerivedStore('activity', {
  sources: ['users', 'projects'],
  derive: (stores) => {
    // Combine data from multiple stores
    return computeUserActivity(stores.users, stores.projects);
  }
});
```

## Complete Example

Here's a complete todo app showcasing kSync's new features:

```typescript
import { z } from 'zod';
import { ksync } from '@klastra/ksync';

// Schema
const TodoSchema = z.object({
  id: z.string(),
  title: z.string(),
  completed: z.boolean(),
  priority: z.enum(['low', 'medium', 'high']),
  createdAt: z.number(),
  updatedAt: z.number(),
  userId: z.string()
});

// Create kSync with database-like queries
const todoApp = ksync.createDrizzleAdapter({
  schema: { todos: TodoSchema },
  syncUrl: 'ws://localhost:3001'
});

// Define events
ksync.defineSchema('todo-created', TodoSchema);
ksync.defineSchema('todo-updated', TodoSchema.partial().extend({ id: z.string() }));
ksync.defineSchema('todo-deleted', z.object({ id: z.string() }));

// Todo service
class TodoService {
  // Create todo
  async createTodo(title: string, priority = 'medium') {
    const todo = {
      id: `todo-${Date.now()}`,
      title,
      completed: false,
      priority,
      createdAt: Date.now(),
      updatedAt: Date.now(),
      userId: 'user-123'
    };
    
    await ksync.send('todo-created', todo);
    return todo;
  }
  
  // Get active todos
  async getActiveTodos() {
    return await todoApp.todos.findMany({
      where: { completed: false },
      orderBy: { priority: 'desc', createdAt: 'desc' }
    });
  }
  
  // Toggle completion
  async toggleTodo(id: string) {
    const todo = await todoApp.todos.findById(id);
    if (todo) {
      await ksync.send('todo-updated', {
        id,
        completed: !todo.completed,
        updatedAt: Date.now()
      });
    }
  }
  
  // Get statistics
  async getStats() {
    const [total, completed, high] = await Promise.all([
      todoApp.todos.count(),
      todoApp.todos.count({ where: { completed: true } }),
      todoApp.todos.count({ where: { priority: 'high' } })
    ]);
    
    return { total, completed, pending: total - completed, high };
  }
}

// Usage
const todoService = new TodoService();

// Create some todos
await todoService.createTodo('Buy groceries', 'high');
await todoService.createTodo('Read documentation', 'medium');

// Query todos
const activeTodos = await todoService.getActiveTodos();
console.log('Active todos:', activeTodos);

// Get stats
const stats = await todoService.getStats();
console.log('Stats:', stats);
```
